// Conferencia: Tech Corp = 20 vagas e 25 interessados; Google = 5 vagas e zero interessados.
// Apos o complemento, os totais de interessados serao 100 e 25, respectivamente.
MATCH (company:COMPANY)
OPTIONAL MATCH (company)<-[:WORKS_FOR]-(admin:USER)
WITH company, collect(admin.email) AS administradores
OPTIONAL MATCH (company)-[:PUBLISHES]->(vacancy:VACANCY)
WITH company, administradores, count(DISTINCT vacancy) AS vagas
OPTIONAL MATCH (candidate:CANDIDATE)-[:`INTERESTED IN`]->(company)
RETURN company.name AS empresa, company.cnpj AS cnpj, administradores, vagas, count(DISTINCT candidate) AS interessados
ORDER BY empresa;
