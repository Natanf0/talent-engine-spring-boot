OPTIONAL MATCH (old:LANGUAGE)
WHERE old.catalogId=spec.id OR toLower(trim(old.name)) IN spec.aliases
WITH spec,old ORDER BY CASE WHEN old.catalogId=spec.id THEN 0 ELSE 1 END,id(old)
WITH spec,head(collect(old)) AS existing
CALL {
  WITH spec,existing WITH spec,existing WHERE existing IS NOT NULL RETURN existing AS lang
  UNION
  WITH spec,existing WITH spec,existing WHERE existing IS NULL
  MERGE (lang:LANGUAGE {normalizedName:toLower(spec.name)}) ON CREATE SET lang.name=spec.name
  RETURN lang
}
SET lang.catalogId=spec.id,lang.catalogName=spec.name
RETURN count(*) AS idiomasDisponiveis;
